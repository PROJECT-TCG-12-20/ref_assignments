4) create table InchesToYards(
    -> value int,
    -> yards int,
    -> feet int,
    -> inches int);
    -> //
create procedure InchesToYards(x int)
    -> begin
    ->  declare yards int;
    ->  declare feet int;
    ->  declare inches int;
    ->  set yards = x/36;
    ->  set feet = (x-36*yards)/12;
    ->  set inches = x-(36*yards)-(12*feet);
    ->  insert into InchesToYards values(x,yards,feet,inches);
    -> end;//
call InchesToYards(124)//
